/// A chat renamed with nothing else changing reaches every library window holding one of its
/// items, across chats as well as in it, as a fresh watch has it.
#[tokio::test(flavor = "multi_thread")]
async fn windows_are_told_when_a_chat_their_items_are_in_is_renamed_alone() {
    let bed = Bed::new(brisk());
    shared(&bed).await;
    bed.store.indexed().await.unwrap();
    let mut w = bed.watcher(64);
    let wants = [
        json!({"kind": "photo", "around": "top", "after": 10}),
        json!({"kind": "all", "around": "top", "after": 3}),
        json!({"kind": "video", "person": "person:2", "around": "top", "after": 10}),
        json!({"kind": "photo", "chat": CLIMBING, "around": "top", "after": 10}),
    ];
    let mut watched = Vec::new();
    for want in &wants {
        watched.push((w.watch("library.watch", want.clone()).await, want.clone()));
    }
    let renamed = Chat { id: CLIMBING.into(), name: Some("Escalada".into()), ..Chat::default() };
    let seq = bed.apply(vec![Change::Chat(renamed)]).await;
    w.settle(seq).await;
    for (watch, want) in &watched {
        let held = &w.windows[watch];
        same(held, &fresh(&bed, want).await, &format!("renamed alone, {want}"));
        let names: Vec<Value> = held.ordered().iter().filter(|row| row["chat"] == CLIMBING).map(|row| row["chat_name"].clone()).collect();
        assert!(!names.is_empty() && names.iter().all(|name| name == "Escalada"), "{want}: {names:?}");
    }
}

/// A library window read again when nothing it holds changed is sent nothing, whichever window it
/// is: total, older and counts go only when they change.
#[tokio::test(flavor = "multi_thread")]
async fn a_window_read_again_with_nothing_it_holds_changed_is_sent_nothing() {
    let bed = Bed::new(brisk());
    shared(&bed).await;
    bed.store.indexed().await.unwrap();
    let mut w = bed.watcher(64);
    let wants = [
        json!({"kind": "photo", "around": "top", "after": 10}),
        json!({"kind": "photo", "around": "top", "after": 1}),
        json!({"kind": "all", "around": "top", "after": 3}),
        json!({"kind": "video", "person": "person:2", "around": "top", "after": 10}),
        json!({"kind": "file", "chat": FAMILIA, "around": "top", "after": 10}),
        json!({"kind": "photo", "text": "praia", "around": "top", "after": 10}),
        json!({"kind": "link", "around": "top", "after": 10}),
    ];
    let mut watched = Vec::new();
    for want in &wants {
        watched.push((w.watch("library.watch", want.clone()).await, want.clone()));
    }
    let reads = bed.feed.reads();
    // A text message with no link in Família's chat, and someone no row names: every window is read
    // again, across chats for any message, Família's for its own, and all of them for anyone, and
    // nothing any of them holds changes.
    bed.apply(vec![
        Change::Message(item(FAMILIA, "t9", 20_000, Some(ANA), "text", Some("até logo"))),
        Change::Person(Person { jids: vec!["351910000009@s.whatsapp.net".into()], name: Some("Zé".into()), ..Person::default() }),
    ])
    .await;
    bed.store.indexed().await.unwrap();
    w.settle(bed.feed.seq() as i64).await;
    w.run_for(crate::feed::tests::ms(200)).await;
    assert!(bed.feed.reads() >= reads + wants.len() as u64, "every window was read again: {} reads", bed.feed.reads() - reads);

        assert_eq!(w.batches.get(watch), None, "{want}: a push with nothing in it changed");
        same(&w.windows[watch], &fresh(&bed, want).await, &format!("{want}"));
    }
}

/// A photo in the open chat reaches its conversation before a library window of photos across
/// chats that was watched first, as a lone message reaches it before the chat list.
#[tokio::test(flavor = "multi_thread")]
async fn a_photo_in_the_open_chat_reaches_its_conversation_before_the_library() {
    let bed = Bed::new(brisk());
    bed.apply(vec![Change::Message(text(OPEN, "first", 1_000, Some(OPEN), "hi"))]).await;
    let mut watcher = bed.watcher(64);
    // Watched first, so it would be read first if the order of watching decided.
    let library = watcher.watch("library.watch", json!({"kind": "photo", "around": "top", "after": 20})).await;
    let messages = watcher.watch("messages.watch", json!({"chat": OPEN, "around": "newest", "before": 50})).await;
    let lone = Message { live: true, text: None, ..photo(OPEN, "lone", 2_000, OPEN) };
    let seq = bed.apply(vec![Change::Message(lone)]).await;
    while watcher.seen < seq || watcher.client.pending() {
        let ready =
            tokio::time::timeout(Duration::from_secs(10), watcher.client.ready()).await.expect("nothing happened");
        if let Ready::Changes(Ok(changes)) = &ready {
            watcher.seen = watcher.seen.max(changes.seq);
        }
        watcher.client.handle(ready).await;
    }
    let mut order = Vec::new();
    while let Ok(line) = watcher.lines.try_recv() {
        let FromJawd::Push(push) = serde_json::from_slice(&line).unwrap() else { panic!("a reply among the pushes") };
        order.push(push.params["watch"].as_str().unwrap().to_string());
    }
    assert_eq!(order, [messages, library]);
}

### Inconsistency

**I1. Windows on a chat being merged away aren't told its messages are leaving unless its unread count moves. Main's code; the library's windows show it too.**
- **Where:**
  - `store/write.rs:1376` (`merge_step` shows only the kept chat's newest);

- **What happens:**
  - The chat merged away is named only when the step's recount of its unread writes a chat event, which happens when the moved messages were unread.
  - With your own messages, or a chat already at the 1,000 cap, nothing names it until the last step removes the chat.
  - Meanwhile its windows keep their old `total`, `older` and `counts`. An anchored window deep in the chat would also keep rows that have moved.
- **Measured:** my probe `audit_windows_on_a_chat_of_yours_being_merged_follow_each_step`, 450 of your photos merged in three steps:

  With unread photos, the same probe passes every step.
- **Why it matters here:** the library claims pushes equal a fresh query, and its counts come from `shelves`, which moves at every step without an event for that chat.

### Unpredictable performance

**P1 (must change). Paging the library makes media fetch every history photo it shows, through the live lane, slowing message arrival and acks past their budgets and growing the WAL past its bound.**
- **Where:**
  - `store/library.rs:399`, where each item read calls `with_media`;
  - `store/read.rs:500`, where that tells media of a copy that's remote;
  - `media/schedule.rs:389`, where the wants are applied in `Lane::Live`;
  - `media/fetch.rs` and `media/schedule.rs`, where every later state is applied in `Lane::Live` too.
- **What happens:**
  - A page of 120 history photos becomes 120 wants in one live transaction, then a download each, and each download's states and outcome are live writes.
  - Live writes go ahead of background work and aren't held back for the checkpointer. So a walk queues hundreds of them ahead of incoming messages' conversions, and grows the WAL without bound.
  - Messages windows do this for the photos they show. A library page shows 120 at once, and a walk shows thousands.
  - On a real account the downloads are paced by the network, so the burst is longer and lower. But the wants are written at once, and history from before linking is often gone from WhatsApp's servers, so it's asked of the phone again.
- **Measured:** Measurements, last table.
  - With media fetching, during back-to-back paging:
    - commit to the interface 32 to 38 ms at p50 and 119 to 189 ms at p99;
    - ack 45 to 130 ms at p99;
    - the WAL up to 281,596 to 298,802 frames (1.2 GB);
    - live writes waiting 83 to 139 ms at p50.
  - With media not told, the same paging leaves commit to the interface at 2.7 ms p50 and 5 to 9 ms p99, and the WAL under 1,200 frames.
- **Fix:** media's work for what windows show must wait behind live messages and keep to the WAL bound. Either:
  - media, being changed now, applies window-driven wants and what follows them in the background lane, with its downloads bounded; or
  - the library tells media only of what an interface has on screen.

  Then show it: lone messages while the library is paged arrive and are acked no slower than without paging (the `JAW_AUDIT_STEADY_LIBRARY` and `JAW_AUDIT_NO_SEEN` switches below make that run). Add a test that window-driven media writes keep to the WAL bound or never ride the live lane.

**P2 (follow-up). History is indexed for search more slowly.**
- **Measured:**
  - The first run's 84,394 history messages were converted in about the same time (6.3 to 10.7 s), but converted and indexed in 126 to 147 s against 57 to 71 s on the base.
  - `jawd bench store`'s 200,000 messages, with no items at all, built in 11.1 to 12.0 s against 9.4 to 9.9 s, and indexed in 22 to 33 s against 17 to 23 s.
  - Live messages during both stayed comparable.
- **Likely causes:**
  - every backlog row now runs a second `INSERT … SELECT` into `library_fts`, even for non-items (`store/write.rs:1018`);
  - items write an event;
  - the writer's merge steps now also merge `library_fts`.
- **Fix, if wanted:**
  - read `shelf` with the backlog's rows and skip the statement for non-items;
  - measure again after the rebase, where live text joins the backlog too.

  No budget names time to searchable.

**P3 (follow-up). Each live write costs about 20 to 45 µs more CPU on a million messages** (Live writes, above). That covers the generated column, the four partial indexes and the triggers' extra statements. It's well inside the live budgets.

**P4 (follow-up). The upgrade leaves jawd 45 MB more resident until it restarts.** 142 MB against 96 to 99 MB, with the footprint 78 to 87 MB, inside budget. It's the same pattern as the budgets audit's P1, and the same fix applies: `temp_store = FILE` for data steps, and `PRAGMA shrink_memory` after.

**P5 (note). One person's items in one chat** filter the whole chat's items of that kind on the index (`store/library.rs:117-121`), since the index can't seek the sender. On the demo's biggest chat, 9,104 photos, that's under a millisecond. A huge group with a rare sender would cost a few milliseconds.
